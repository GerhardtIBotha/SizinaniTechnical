-- ===== 1. Database & tables =====

CREATE TABLE dbo.RegisteredVehicle(
	RegisteredVehicleID INT IDENTITY PRIMARY KEY,
	Type NVARCHAR (50) NOT NULL,
	RegistrationNr NVARCHAR (10) NOT NULL,
	Modal NVARCHAR (50) NOT NULL,
	Weight DECIMAL (7, 2) NOT NULL
);

CREATE TABLE dbo.Contractor (
	ContractorID INT IDENTITY PRIMARY KEY,
	ContractorFirstName NVARCHAR (50) NOT NULL,
	ContractorLastName NVARCHAR (50)NOT NULL,
	ContractorEmail NVARCHAR (319)NOT NULL,
	ContractorPhone INT NOT NULL,
);

CREATE TABLE dbo.ContractorHasRegisteredVehicle(
	ContractorHasRegisteredVehicleID INT IDENTITY PRIMARY KEY,
	ContractorID INT NOT NULL REFERENCES dbo.Contractor(ContractorID),
	RegisteredVehicleID INT NOT NULL REFERENCES dbo.RegisteredVehicle(RegisteredVehicleID)
);

-- ===== 3. Stored procedures =====

CREATE PROCEDURE dbo.usp_GetVehiclesByContractor
		@ContractorID INT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT  rv.RegisteredVehicleID,
            rv.Type,
            rv.RegistrationNr,
            rv.Modal,
            rv.Weight
    FROM    dbo.ContractorHasRegisteredVehicle chrv
    JOIN    dbo.RegisteredVehicle rv
            ON rv.RegisteredVehicleID = chrv.RegisteredVehicleID
    WHERE   chrv.ContractorID = @ContractorID
    ORDER BY rv.Type, rv.RegistrationNr;
END
GO
--------------------------------
CREATE PROCEDURE dbo.usp_GetContractors
AS
BEGIN
    SET NOCOUNT ON;
    SELECT  ContractorID,
            ContractorFirstName + ' ' + ContractorLastName AS ContractorName
    FROM    dbo.Contractor
    ORDER BY ContractorLastName, ContractorFirstName;
END
GO
